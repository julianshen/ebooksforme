第 6 章：GOVERN 層 — 治理與合規

6.1 問題：如何管理數百個 Agent？

當你的平台上有數百個 Agent 同時運行時，你會面臨一系列治理挑戰：

  • 身份管理：如何確保每個 Agent 的身份真實可信？
  • 能力發現：如何知道每個 Agent 能做什麼？
  • 行為審計：如何追蹤每個 Agent 的操作？
  • 政策執行：如何確保 Agent 遵守安全政策？
  • 合規報告：如何證明系統符合法規要求？

GOVERN 層的目標是讓 Agent 平台可管理、可審計、可合規。

6.2 Agent Registry

Agent Registry 是 Agent 的「黃頁」，記錄所有可用 Agent 的資訊。

6.2.3 發現機制

func (r *Registry) Discover(capability string) ([]*AgentInfo, error) {
    // 掃描所有 Agent
    keys, _ := r.kv.Keys()

    var agents []*AgentInfo
    for _, key := range keys {
        entry, _ := r.kv.Get(key)
        var info AgentInfo
        json.Unmarshal(entry.Value(), &info)

        // 檢查能力匹配
        if contains(info.Capabilities, capability) {
            // 檢查健康狀態
            if time.Since(info.LastSeen) < 2*time.Minute {
                agents = append(agents, &info)
            }
        }
    }

    return agents, nil
}

6.3 Agent Identity

6.3.1 SPIFFE/SPIRE 整合

type IdentityManager struct {
    spireClient *spire.Client
}

func (im *IdentityManager) IssueIdentity(agentID string) (*Identity, error) {
    // 1. 驗證 Agent 註冊資訊
    info, err := registry.Get(agentID)
    if err != nil {
        return nil, fmt.Errorf("agent not registered: %w", err)
    }

    // 2. 請求 SPIRE 簽發 SVID
    svid, err := im.spireClient.FetchX509SVID(
        spiffeid.Must("spiffe://cluster.local/ns/manus-platform/sa/" + info.Name),
    )

    // 3. 返回身份
    return &Identity{
        SPIFFEID: svid.ID.String(),
        Certificate: svid.Certificates,
        PrivateKey: svid.PrivateKey,
        ExpiresAt: svid.ExpiresAt,
    }, nil
}

6.4 審計追蹤

6.4.1 審計事件類型

type AuditEvent struct {
    Timestamp   time.Time         `json:"timestamp"`
    EventType   string            `json:"event_type"`
    Actor       string            `json:"actor"`        // Agent ID 或使用者 ID
    Action      string            `json:"action"`
    Resource    string            `json:"resource"`
    Status      string            `json:"status"`       // success / failure
    Details     map[string]interface{} `json:"details"`
    RequestID   string            `json:"request_id"`
}

// 事件類型
const (
    EventTaskCreated     = "task.created"
    EventTaskAssigned    = "task.assigned"
    EventToolCalled      = "tool.called"
    EventToolDenied      = "tool.denied"
    EventAgentRegistered = "agent.registered"
    EventAgentDeregistered = "agent.deregistered"
    EventPolicyViolated  = "policy.violated"
)

6.4.2 實作

type Auditor struct {
    js jetstream.JetStream
}

func (a *Auditor) Log(ctx context.Context, event *AuditEvent) error {
    // 1. 序列化事件
    data, _ := json.Marshal(event)

    // 2. 發布到 JetStream
    // 使用 Subject 分類，便於訂閱和分析
    subject := fmt.Sprintf("audit.%s.%s", event.Actor, event.Action)

    _, err := a.js.Publish(ctx, subject, data,
        jetstream.WithMsgID(event.RequestID),  // 去重
    )

    return err
}

// 使用範例
auditor.Log(ctx, &AuditEvent{
    EventType: EventToolCalled,
    Actor:     "coding-agent-1",
    Action:    "file.read",
    Resource:  "/app/main.go",
    Status:    "success",
    Details: map[string]interface{}{
        "file_size": 1024,
        "read_lines": 50,
    },
    RequestID: reqID,
})

6.4.3 審計日誌分析

func (a *Auditor) Analyze(ctx context.Context, timeframe time.Duration) (*AuditReport, error) {
    // 1. 讀取時間範圍內的所有事件
    events := a.queryEvents(ctx, timeframe)

    // 2. 統計
    report := &AuditReport{
        TotalEvents: len(events),
        EventTypes:  make(map[string]int),
        Actors:      make(map[string]int),
        Violations:  []AuditEvent{},
    }

    for _, event := range events {
        report.EventTypes[event.EventType]++
        report.Actors[event.Actor]++

        if event.EventType == EventPolicyViolated {
            report.Violations = append(report.Violations, event)
        }
    }

    // 3. 檢測異常模式
    report.Anomalies = a.detectAnomalies(events)

    return report, nil
}

6.5 政策執行

6.5.1 政策類型

type Policy struct {
    ID          string      `json:"id"`
    Name        string      `json:"name"`
    Description string      `json:"description"`
    Type        PolicyType  `json:"type"`
    Rules       []Rule      `json:"rules"`
    Priority    int         `json:"priority"`
    Enabled     bool        `json:"enabled"`
}

type PolicyType string

const (
    PolicyTypeNetwork   PolicyType = "network"   // 網路政策
    PolicyTypeTool      PolicyType = "tool"      // 工具政策
    PolicyTypeResource  PolicyType = "resource"  // 資源政策
    PolicyTypeContent   PolicyType = "content"   // 內容政策
)

6.5.2 工具政策範例

# 工具使用政策
apiVersion: policy.manus.io/v1
kind: ToolPolicy
metadata:
  name: coding-agent-tools
spec:
  agentSelector:
    matchLabels:
      agent-type: coding
  rules:
  - tool: file.read
    effect: allow
    conditions:
    - path: "/app/*"          # 僅允許讀取 /app 目錄
  - tool: file.write
    effect: allow
    conditions:
    - path: "/app/*"
    - maxSize: "10MB"         # 單次寫入限制
  - tool: network.request
    effect: deny               # Coding Agent 禁止網路請求
  - tool: code.execute
    effect: allow
    conditions:
    - timeout: "30s"           # 執行逾時
    - memory: "1Gi"            # 記憶體限制

6.5.3 政策執行引擎

func (e *PolicyEngine) Evaluate(ctx context.Context, agentID string, action Action) (*Decision, error) {
    // 1. 獲取 Agent 資訊
    agent, _ := e.registry.Get(agentID)

    // 2. 載入適用的政策
    policies := e.loadPolicies(agent)

    // 3. 按優先級排序
    sort.Slice(policies, func(i, j int) bool {
        return policies[i].Priority > policies[j].Priority
    })

    // 4. 逐條評估
    for _, policy := range policies {
        for _, rule := range policy.Rules {
            if rule.Matches(action) {
                return &Decision{
                    Allowed: rule.Effect == "allow",
                    Reason:  rule.Description,
                    Policy:  policy.ID,
                }, nil
            }
        }
    }

    // 5. 預設拒絕
    return &Decision{Allowed: false, Reason: "no matching rule"}, nil
}

6.6 合規報告

6.6.1 報告類型

type ComplianceReport struct {
    Period      time.TimeRange    `json:"period"`
    GeneratedAt time.Time         `json:"generated_at"`

    // 存取控制
    AccessControl *AccessControlReport `json:"access_control"`

    // 資料保護
    DataProtection *DataProtectionReport `json:"data_protection"`

    // 審計完整性
    AuditIntegrity *AuditIntegrityReport `json:"audit_integrity"`
}

6.6.2 自動生成

func (g *ReportGenerator) Generate(ctx context.Context, period time.TimeRange) (*ComplianceReport, error) {
    // 1. 讀取審計日誌
    events := g.auditor.Query(ctx, period)

    // 2. 分析存取控制
    accessReport := g.analyzeAccessControl(events)

    // 3. 分析資料保護
    dataReport := g.analyzeDataProtection(events)

    // 4. 驗證審計完整性
    integrityReport := g.verifyAuditIntegrity(events)

    return &ComplianceReport{
        Period:         period,
        GeneratedAt:    time.Now(),
        AccessControl:  accessReport,
        DataProtection: dataReport,
        AuditIntegrity: integrityReport,
    }, nil
}

6.7 常見陷阱

陷阱 1：審計日誌不完整

// 錯誤：僅記錄成功操作
if err == nil {
    auditor.Log(ctx, event)  // 失敗操作丟失！
}

// 正確：記錄所有操作，包括失敗
auditor.Log(ctx, event)  // 無論成功或失敗都記錄
// 在 Details 中標註狀態
event.Status = map[bool]string{true: "success", false: "failure"}[err == nil]

陷阱 2：身份輪換中斷服務

// 錯誤：SVID 過期後才更新
if time.Now().After(svid.ExpiresAt) {
    svid = im.IssueIdentity(agentID)  // 服務中斷！
}

// 正確：提前輪換
if time.Until(svid.ExpiresAt) < time.Hour {
    svid = im.IssueIdentity(agentID)  // 在過期前更新
}

陷阱 3：政策衝突

# 錯誤：政策 A 允許，政策 B 拒絕，結果不明確
# policy-a.yaml
rules:
- tool: file.read
  effect: allow

# policy-b.yaml
rules:
- tool: file.read
  effect: deny

# 正確：明確優先級
# policy-a.yaml (priority: 100)
# policy-b.yaml (priority: 50)
# 結果：政策 A 優先，允許

6.8 總結

GOVERN 層讓 Agent 平台可管理、可審計、可合規：

  1. Agent Registry：統一管理 Agent 註冊和發現
  2. Identity：SPIFFE/SPIRE 提供加密身份
  3. Auditor：完整記錄所有操作
  4. Policy Engine：動態執行安全政策
  5. Compliance：自動生成合規報告

關鍵原則：
- 所有操作必須可審計
- 身份必須可驗證
- 政策必須可動態更新
- 合規必須可自動化
