第 4 章：SCALE 層 — 讓 Agent 可靠運行

4.1 問題：Agent 在生產環境會遇到什麼？

開發環境和生產環境是兩個完全不同的世界。

在開發環境，Agent 跑在你的筆記本電腦上，網路穩定，資源充足，你可以隨時查看日誌。在生產環境，Agent 跑在 Kubernetes 集群中，面臨：

  • 節點故障：物理機器或 VM 突然當機
  • 網路分區：Agent 與 Orchestrator 之間的網路中斷
  • 資源耗盡：記憶體洩漏導致 OOMKilled
  • 依賴故障：LLM API 逾時或返回錯誤
  • 狀態丟失：Pod 重啟後所有上下文消失

SCALE 層的目標是讓 Agent 在這些情況下依然可靠運行。

4.2 Agent Runtime

Agent Runtime 是 Agent 的執行環境。它需要處理：

4.2.1 生命週期管理

每個狀態轉換都需要持久化，確保故障後可恢復。

4.2.2 健康檢查

type HealthChecker struct {
    interval time.Duration
    timeout  time.Duration
}

func (h *HealthChecker) Check(agent *Agent) HealthStatus {
    // 1. 檢查程序是否存活
    if !agent.IsAlive() {
        return HealthStatusDead
    }

    // 2. 檢查心跳
    if time.Since(agent.LastHeartbeat) > h.timeout {
        return HealthStatusUnresponsive
    }

    // 3. 檢查資源使用
    if agent.MemoryUsage > 0.9 {
        return HealthStatusResourceExhausted
    }

    return HealthStatusHealthy
}

4.2.3 自動恢復

func (r *Runtime) HandleFailure(agent *Agent, status HealthStatus) {
    switch status {
    case HealthStatusDead:
        // 重新啟動 Agent
        r.RestartAgent(agent)

    case HealthStatusUnresponsive:
        // 先嘗試優雅重啟
        if err := r.GracefulRestart(agent); err != nil {
            // 失敗則強制重啟
            r.ForceRestart(agent)
        }

    case HealthStatusResourceExhausted:
        // 遷移到資源充足的節點
        r.MigrateAgent(agent)
    }
}

type SessionManager struct {
    hotStore  *redis.Client      // 活躍 Session
    warmStore jetstream.KeyValue  // 近期 Session
    coldStore *s3.Client          // 歷史 Session
}

func (sm *SessionManager) GetSession(ctx context.Context, id string) (*Session, error) {
    // 1. 嘗試從 Hot Store 讀取
    if session, err := sm.hotStore.Get(ctx, id); err == nil {
        return session, nil
    }

    // 2. 嘗試從 Warm Store 讀取
    if entry, err := sm.warmStore.Get(ctx, id); err == nil {
        session := deserialize(entry.Value())
        // 提升到 Hot Store
        sm.hotStore.Set(ctx, id, session)
        return session, nil
    }

    // 3. 從 Cold Store 讀取
    return sm.coldStore.Get(ctx, id)
}

func (sm *SessionManager) SaveSession(ctx context.Context, session *Session) error {
    // 儲存到 Hot Store
    if err := sm.hotStore.Set(ctx, session.ID, session); err != nil {
        return err
    }

    // 異步儲存到 Warm Store
    go func() {
        data, _ := json.Marshal(session)
        sm.warmStore.Put(ctx, session.ID, data)
    }()

    return nil
}

4.4 Memory Bank

Memory Bank 是 Agent 的長期記憶。與 Session 不同，它跨 Session 持久存在。

4.4.1 記憶類型

type MemoryType string

const (
    MemoryTypePreference  MemoryType = "preference"  // 使用者偏好
    MemoryTypeFact        MemoryType = "fact"        // 學到的事實
    MemoryTypeInteraction MemoryType = "interaction" // 互動摘要
    MemoryTypeContext     MemoryType = "context"     // 應跨 Session 的上下文
)

type Memory struct {
    ID        string     `json:"id"`
    UserID    string     `json:"user_id"`
    Type      MemoryType `json:"type"`
    Content   string     `json:"content"`
    Keywords  []string   `json:"keywords"`
    Importance float32   `json:"importance"` // 0-1
    CreatedAt time.Time  `json:"created_at"`
    UpdatedAt time.Time  `json:"updated_at"`
}

4.4.2 檢索機制

func (mb *MemoryBank) Retrieve(ctx context.Context, userID string, query string, limit int) ([]*Memory, error) {
    // 1. 關鍵字匹配（簡易）
    keywords := extractKeywords(query)

    // 2. 向量相似度（生產環境）
    queryEmbedding := mb.encoder.Encode(query)

    // 3. 組合檢索
    memories := mb.db.Query(ctx, userID, keywords, queryEmbedding, limit)

    // 4. 按重要性排序
    sort.Slice(memories, func(i, j int) bool {
        return memories[i].Importance > memories[j].Importance
    })

    return memories, nil
}

4.4.3 記憶壓縮

長期記憶會不斷增長。我們需要壓縮機制：

func (mb *MemoryBank) Compact(ctx context.Context, userID string) error {
    // 1. 獲取該使用者的所有記憶
    memories := mb.db.GetByUser(ctx, userID)

    // 2. 按時間分組
    groups := groupByTime(memories, time.Hour*24)

    // 3. 每組生成摘要
    for _, group := range groups {
        summary := mb.llm.Summarize(group)

        // 4. 儲存摘要，刪除原始記憶
        mb.db.Save(ctx, &Memory{
            UserID:   userID,
            Type:     MemoryTypeInteraction,
            Content:  summary,
            Importance: calculateImportance(group),
        })

        mb.db.DeleteBatch(ctx, group)
    }

    return nil
}

4.5 Sandbox 控制器

Sandbox 控制器負責動態建立和銷毀 Agent 的執行環境。

4.5.1 設計

type SandboxController struct {
    k8sClient kubernetes.Interface
    factory   *SandboxFactory
    scheduler *SandboxScheduler
}

func (sc *SandboxController) CreateSandbox(ctx context.Context, spec *SandboxSpec) (*Sandbox, error) {
    // 1. 選擇隔離等級
    isolation := sc.selectIsolation(spec.AgentType, spec.IsolationLevel)

    // 2. 選擇節點
    node := sc.scheduler.SelectNode(spec.Resources, isolation)

    // 3. 建立 Pod
    pod := sc.factory.CreatePod(spec, isolation, node)

    // 4. 等待就緒
    if err := sc.waitForReady(ctx, pod); err != nil {
        sc.cleanup(pod)
        return nil, err
    }

    return &Sandbox{Pod: pod}, nil
}

4.5.2 隔離等級選擇

func (sc *SandboxController) selectIsolation(agentType string, requested IsolationLevel) IsolationConfig {
    // 預設隔離策略
    defaults := map[string]IsolationConfig{
        "coding":   {Runtime: "runc", NetworkPolicy: "restricted"},
        "browser":  {Runtime: "gvisor", NetworkPolicy: "permissive"},
        "data":     {Runtime: "runc", NetworkPolicy: "restricted"},
        "review":   {Runtime: "runc", NetworkPolicy: "restricted"},
    }

    config := defaults[agentType]

    // 使用者可覆蓋
    if requested != "" {
        config.Runtime = string(requested)
    }

    return config
}

4.6 常見陷阱

陷阱 1：Session 狀態過大

// 錯誤：儲存所有對話歷史
session.History = append(session.History, turn)  // 無限增長

// 正確：定期摘要，只保留近期詳細記錄
if len(session.History) > 100 {
    summary := summarize(session.History[:50])
    session.History = append([]Turn{summary}, session.History[50:]...)
}

陷阱 2：Memory Bank 檢索過慢

// 錯誤：線性掃描所有記憶
for _, memory := range allMemories {
    if match(memory, query) {
        results = append(results, memory)
    }
}

// 正確：使用向量資料庫
memories := pgvector.Search(queryEmbedding, limit)

陷阱 3：Sandbox 資源洩漏

// 錯誤：建立後忘記銷毀
sandbox := controller.CreateSandbox(spec)
// ... 使用 ...
// 忘記銷毀！

// 正確：使用 defer 或 context
func runInSandbox(ctx context.Context, spec *SandboxSpec, fn func()) error {
    sandbox, err := controller.CreateSandbox(ctx, spec)
    if err != nil {
        return err
    }
    defer controller.DestroySandbox(ctx, sandbox)  // 確保銷毀

    fn()
    return nil
}

4.7 總結

SCALE 層的核心是讓 Agent 在生產環境可靠運行：

  1. Agent Runtime：管理生命週期、健康檢查、自動恢復
  2. Session Manager：分層儲存策略（Hot/Warm/Cold）
  3. Memory Bank：跨 Session 長期記憶，支援檢索和壓縮
  4. Sandbox Controller：動態建立隔離環境

關鍵原則：
- 所有狀態都要持久化
- 故障是自動恢復，不是人工介入
- 資源使用要有上限和清理機制
