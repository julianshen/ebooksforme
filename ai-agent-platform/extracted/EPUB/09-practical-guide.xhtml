9.3.3 建構基礎 Agent

// pkg/agent/base.go
package agent

import (
    "context"
    "encoding/json"
    "fmt"
    "time"

    "github.com/nats-io/nats.go"
    "github.com/nats-io/nats.go/jetstream"
)

// BaseAgent 是所有 Agent 的基礎
type BaseAgent struct {
    ID       string
    Name     string
    Type     string
    NC       *nats.Conn
    JS       jetstream.JetStream
    KV       jetstream.KeyValue
    Status   Status
    Capabilities []string
}

// Status 定義 Agent 狀態
type Status string

const (
    StatusIdle      Status = "idle"
    StatusBusy      Status = "busy"
    StatusError     Status = "error"
    StatusShutdown  Status = "shutdown"
)

// NewBaseAgent 建立基礎 Agent
func NewBaseAgent(id, name, agentType string, nc *nats.Conn) (*BaseAgent, error) {
    js, _ := jetstream.New(nc)
    kv, _ := js.CreateKeyValue(context.Background(), jetstream.KeyValueConfig{
        Bucket: "AGENT_REGISTRY",
    })

    return &BaseAgent{
        ID:     id,
        Name:   name,
        Type:   agentType,
        NC:     nc,
        JS:     js,
        KV:     kv,
        Status: StatusIdle,
    }, nil
}

// Register 向 Registry 註冊
func (a *BaseAgent) Register(ctx context.Context) error {
    info := AgentInfo{
        ID:        a.ID,
        Name:      a.Name,
        Type:      a.Type,
        Status:    string(a.Status),
        Endpoint:  fmt.Sprintf("agents.%s.rpc", a.ID),
        Capabilities: a.Capabilities,
        RegisteredAt: time.Now(),
        LastSeen:  time.Now(),
    }

    data, _ := json.Marshal(info)
    _, err := a.KV.Put(ctx, a.ID, data)
    return err
}

// StartHeartbeat 開始發送心跳
func (a *BaseAgent) StartHeartbeat(ctx context.Context, interval time.Duration) {
    ticker := time.NewTicker(interval)
    go func() {
        for {
            select {
            case <-ticker.C:
                a.sendHeartbeat(ctx)
            case <-ctx.Done():
                ticker.Stop()
                return
            }
        }
    }()
}

func (a *BaseAgent) sendHeartbeat(ctx context.Context) {
    heartbeat := Heartbeat{
        AgentID:   a.ID,
        Status:    string(a.Status),
        Timestamp: time.Now(),
    }

    data, _ := json.Marshal(heartbeat)
    a.NC.Publish(fmt.Sprintf("agents.%s.heartbeat", a.ID), data)
}

9.4 第三步：建構 Orchestrator

9.4.1 Python ADK Orchestrator

# python-adk/orchestrator/main.py
from google.adk import Agent, LlmAgent, Runner
from google.adk.tools import AgentTool
import asyncio
import nats

class ManusOrchestrator:
    def __init__(self):
        self.nc = None
        self.js = None
        self.agents = {}

    async def connect(self):
        """連線到 NATS"""
        self.nc = await nats.connect("nats://localhost:4222")
        self.js = self.nc.jetstream()

    async def discover_agents(self):
        """發現可用 Agent"""
        kv = await self.js.key_value(bucket="AGENT_REGISTRY")

        # 獲取所有 Agent
        keys = await kv.keys()
        for key in keys:
            entry = await kv.get(key)
            info = json.loads(entry.value)

            # 檢查健康狀態
            if self.is_healthy(info):
                self.agents[info['name']] = info

    def is_healthy(self, info):
        """檢查 Agent 是否健康"""
        last_seen = datetime.fromisoformat(info['last_seen'])
        return datetime.now() - last_seen < timedelta(minutes=2)

    async def handle_task(self, task):
        """處理新任務"""
        # 1. 規劃任務
        plan = await self.plan_task(task)

        # 2. 執行任務圖
        for step in plan.steps:
            agent = self.select_agent(step)
            result = await self.execute_step(agent, step)

            if result.status == "failed":
                # 重試或回退
                result = await self.handle_failure(step, result)

        # 3. 彙總結果
        return self.aggregate_results(plan)

    async def plan_task(self, task):
        """使用 LLM 規劃任務"""
        planner = LlmAgent(
            model="gemini-2.0",
            instruction="""你是一個任務規劃器。請將使用者請求拆解為可執行的子任務。

可用 Agent 類型：
- coding: 程式碼相關任務
- browser: 網頁互動任務
- data: 資料分析任務
- review: 審查和驗證任務

輸出格式：
1. [agent_type] 具體任務描述
2. [agent_type] 具體任務描述
..."""
        )

        response = await planner.run(task)
        return self.parse_plan(response)

    def select_agent(self, step):
        """選擇最合適的 Agent"""
        candidates = [
            agent for agent in self.agents.values()
            if step.agent_type in agent['capabilities']
        ]

        # 選擇負載最低的
        return min(candidates, key=lambda a: a.get('load', 0))

    async def execute_step(self, agent, step):
        """執行單個步驟"""
        # 通過 NATS RPC 呼叫 Agent
        request = {
            "task": step.description,
            "input": step.input,
            "timeout": step.timeout
        }

        response = await self.nc.request(
            agent['endpoint'],
            json.dumps(request).encode(),
            timeout=30
        )

        return json.loads(response.data)

# 啟動 Orchestrator
async def main():
    orchestrator = ManusOrchestrator()
    await orchestrator.connect()
    await orchestrator.discover_agents()

    # 訂閱新任務
    sub = await orchestrator.js.subscribe("tasks.new")
    async for msg in sub:
        task = json.loads(msg.data)
        result = await orchestrator.handle_task(task)
        await msg.ack()

if __name__ == "__main__":
    asyncio.run(main())

9.6.1 端到端測試

// tests/e2e_test.go
package tests

import (
    "context"
    "testing"
    "time"

    "github.com/nats-io/nats.go"
)

func TestEndToEnd(t *testing.T) {
    // 1. 連線到 NATS
    nc, _ := nats.Connect(nats.DefaultURL)
    defer nc.Close()

    // 2. 發布測試任務
    task := Task{
        ID:      "test-task-001",
        Message: &Message{
            Role: "user",
            Parts: []Part{{
                Type:    "text",
                Content: "分析這個 Go 函數的品質",
            }},
        },
    }

    taskJSON, _ := json.Marshal(task)
    nc.Publish("tasks.new", taskJSON)

    // 3. 等待結果
    sub, _ := nc.SubscribeSync("tasks.completed.test-task-001")
    msg, err := sub.NextMsgWithContext(ctx)
    if err != nil {
        t.Fatalf("任務超時: %v", err)
    }

    // 4. 驗證結果
    var result Task
    json.Unmarshal(msg.Data, &result)

    if result.Status != TaskStatusCompleted {
        t.Fatalf("任務失敗: %s", result.Status)
    }

    t.Logf("任務完成，耗時: %v", result.Duration)
}

9.7 常見問題

Q: Agent 啟動後無法註冊到 Registry

檢查清單：
1. NATS 連線是否正常？
2. KV Store "AGENT_REGISTRY" 是否已建立？
3. Agent 的 ServiceAccount 是否有權限？

# 檢查 NATS 連線
nats server check connection

# 檢查 KV Store
nats kv info AGENT_REGISTRY

# 檢查 Agent 日誌
kubectl logs -l app=coding-agent -n manus-platform

Q: Orchestrator 無法發現 Agent

檢查清單：
1. Agent 是否發送心跳？
2. 心跳間隔是否過長？
3. KV Store 中的時間戳是否正確？

# 監聽心跳
nats sub "agents.*.heartbeat"

# 檢查 KV 內容
nats kv get AGENT_REGISTRY coding-agent-1

Q: 任務執行逾時

檢查清單：
1. Agent 是否忙碌？
2. 任務超時配置是否合理？
3. Agent 資源是否充足？

# 檢查 Agent 狀態
kubectl top pods -l app=coding-agent -n manus-platform

# 檢查 Agent 日誌
kubectl logs coding-agent-xxx -n manus-platform

9.8 總結

本章帶你從零開始建構了一個生產級 Agent 平台：

  1. 環境準備：K3s + NATS + PostgreSQL + Redis
  2. NATS 部署：JetStream + KV Store
  3. Agent Runtime：Go 基礎 Agent 實作
  4. Orchestrator：Python ADK 任務調度
  5. K8s 部署：Deployment + ServiceAccount + Security
  6. 驗證測試：端到端 + 負載測試

關鍵原則：
- 先讓基礎設施跑起來
- 再逐步增加複雜度
- 每個階段都有明確的驗證標準
