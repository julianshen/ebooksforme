10.1 需求分析

打造一個智能客服系統，具備：

  • 🎯 意圖識別（查訂單 / 退貨 / 技術支援 / 轉人工）
  • 📦 查詢訂單狀態
  • 🔄 處理退貨申請
  • 🛠️ 技術問題診斷
  • 👤 高風險操作轉人工

10.2 系統架構

[用戶] -> [意圖識別] -> [條件分支]
                    |
        ┌──────────┼──────────┬──────────┐
        ▼          ▼          ▼          ▼
    [查訂單]   [退貨]    [技術支援]  [轉人工]
        |          |          |          |
        └──────────┴──────────┴──────────┘
                    |
                    ▼
              [總結回覆] -> [用戶]

10.3 完整程式碼

from typing import TypedDict, Annotated
from langgraph.graph import StateGraph, START, END
from langgraph.prebuilt import ToolNode
from langgraph.checkpoint.memory import MemorySaver
from langchain_openai import ChatOpenAI
from langchain_core.tools import tool

# State
class CSState(TypedDict):
    messages: Annotated[list, add_messages]
    intent: str
    order_id: str
    requires_human: bool

# Tools
@tool
def get_order_status(order_id: str) -> str:
    """查詢訂單狀態"""
    return f"訂單 {order_id} 狀態：已出貨，預計明天送達"

@tool
def process_return(order_id: str, reason: str) -> str:
    """處理退貨"""
    return f"退貨申請已提交：{order_id}，原因：{reason}"

# Nodes
def intent_classifier(state: CSState):
    """識別用戶意圖"""
    llm = ChatOpenAI()
    response = llm.invoke([
        ("system", "Classify intent: order_query, return, tech_support, human"),
        ("human", state["messages"][-1].content)
    ])
    intent = response.content.strip()
    return {"intent": intent}

def order_handler(state: CSState):
    """處理查訂單"""
    return {"messages": ["我來幫您查詢訂單..."]}

def return_handler(state: CSState):
    """處理退貨"""
    return {"messages": ["我來協助您辦理退貨..."]}

def tech_handler(state: CSState):
    """技術支援"""
    return {"messages": ["請描述您的技術問題..."]}

def human_handoff(state: CSState):
    """轉人工"""
    return {
        "messages": ["為您轉接專員..."],
        "requires_human": True
    }

def summarize(state: CSState):
    """總結回覆"""
    return {"messages": ["還有其他可以幫您的嗎？"]}

# 條件路由
def route_intent(state: CSState):
    intent_map = {
        "order_query": "order_handler",
        "return": "return_handler",
        "tech_support": "tech_handler",
        "human": "human_handoff"
    }
    return intent_map.get(state["intent"], "human_handoff")

# 建圖
graph = StateGraph(CSState)
graph.add_node("classify", intent_classifier)
graph.add_node("order_handler", order_handler)
graph.add_node("return_handler", return_handler)
graph.add_node("tech_handler", tech_handler)
graph.add_node("human_handoff", human_handoff)
graph.add_node("summarize", summarize)

graph.add_edge(START, "classify")
graph.add_conditional_edges("classify", route_intent)
graph.add_edge("order_handler", "summarize")
graph.add_edge("return_handler", "summarize")
graph.add_edge("tech_handler", "summarize")
graph.add_edge("human_handoff", END)
graph.add_edge("summarize", END)

# 編譯（帶 Checkpoint）
app = graph.compile(checkpointer=MemorySaver())

# 執行
config = {"configurable": {"thread_id": "user-123"}}
result = app.invoke({"messages": [("human", "我的訂單到哪了？")]}, config)
print(result["messages"][-1].content)

10.4 部署到生產環境

# 使用 LangGraph Platform 部署
from langgraph_sdk import get_client

client = get_client(url="https://api.langgraph.com")

# 上傳圖
assistant = client.assistants.create(
    graph_id="customer-service",
    config={"configurable": {"model": "gpt-4"}}
)

# 建立對話執行緒
thread = client.threads.create()

# 發送訊息
response = client.runs.create(
    thread_id=thread["thread_id"],
    assistant_id=assistant["assistant_id"],
    input={"messages": [{"role": "user", "content": "查訂單"}]}
)
