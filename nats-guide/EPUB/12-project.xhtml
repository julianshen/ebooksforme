12.4 核心程式碼

12.4.1 追蹤服務

package tracking

import (
    "encoding/json"
    "time"
    "github.com/nats-io/nats.go"
)

type Location struct {
    PackageID string    `json:"package_id"`
    Lat       float64   `json:"lat"`
    Lng       float64   `json:"lng"`
    Timestamp time.Time `json:"timestamp"`
}

type TrackingService struct {
    nc *nats.Conn
    js nats.JetStreamContext
}

func NewTrackingService(nc *nats.Conn) (*TrackingService, error) {
    js, err := nc.JetStream()
    if err != nil {
        return nil, err
    }

    // 建立位置 Stream
    js.AddStream(&nats.StreamConfig{
        Name:     "LOCATIONS",
        Subjects: []string{"tracking.location.*"},
        Storage:  nats.FileStorage,
        MaxAge:   7 * 24 * time.Hour,
    })

    return &TrackingService{nc: nc, js: js}, nil
}

func (s *TrackingService) UpdateLocation(loc Location) error {
    data, _ := json.Marshal(loc)
    subject := fmt.Sprintf("tracking.location.%s", loc.PackageID)
    _, err := s.js.Publish(subject, data)
    return err
}

func (s *TrackingService) SubscribeLocation(packageID string) (*nats.Subscription, error) {
    return s.js.Subscribe(
        fmt.Sprintf("tracking.location.%s", packageID),
        func(msg *nats.Msg) {
            var loc Location
            json.Unmarshal(msg.Data, &loc)
            // 更新資料庫、發送通知...
        },
        nats.Durable("tracker-"+packageID),
    )
}

12.4.2 調度服務

package dispatch

import (
    "encoding/json"
    "github.com/nats-io/nats.go"
)

type DispatchService struct {
    nc *nats.Conn
}

func (s *DispatchService) Start() {
    // 訂閱新訂單
    s.nc.Subscribe("orders.created", func(msg *nats.Msg) {
        var order Order
        json.Unmarshal(msg.Data, &order)

        // 找到最佳司機
        driver := s.findBestDriver(order)

        // 發送分配指令
        s.nc.Publish("dispatch.assign", json.Marshal(map[string]interface{}{
            "order_id":  order.ID,
            "driver_id": driver.ID,
            "route":     s.calculateRoute(order, driver),
        }))
    })

    // 訂閱異常事件
    s.nc.Subscribe("orders.exception", func(msg *nats.Msg) {
        var event ExceptionEvent
        json.Unmarshal(msg.Data, &event)

        // 重新分配
        if event.Type == "delivery_failed" {
            s.reassignOrder(event.OrderID)
        }
    })
}

func (s *DispatchService) findBestDriver(order Order) Driver {
    // 使用 Request/Reply 查詢司機狀態
    resp, _ := s.nc.Request("drivers.available",
        []byte(order.PickupLocation), 2*time.Second)

    var drivers []Driver
    json.Unmarshal(resp.Data, &drivers)

    // 選擇最近且空閒的司機
    return s.selectOptimal(drivers, order)
}

12.4.3 通知服務

package notification

import (
    "encoding/json"
    "github.com/nats-io/nats.go"
)

type NotificationService struct {
    nc *nats.Conn
}

func (s *NotificationService) Start() {
    // 訂閱所有需要通知的事件
    s.nc.Subscribe("orders.>", func(msg *nats.Msg) {
        event := msg.Subject // orders.created, orders.delivered, etc.
        
        switch event {
        case "orders.created":
            s.sendNotification(msg, "訂單已建立")
        case "orders.picked_up":
            s.sendNotification(msg, "包裹已取件")
        case "orders.in_transit":
            s.sendNotification(msg, "包裹運送中")
        case "orders.delivered":
            s.sendNotification(msg, "包裹已送達")
        }
    })
}

func (s *NotificationService) sendNotification(msg *nats.Msg, template string) {
    var order Order
    json.Unmarshal(msg.Data, &order)

    // 發送推播通知
    s.nc.Publish("notifications.push", json.Marshal(PushNotification{
        UserID:  order.CustomerID,
        Title:   "物流更新",
        Body:    fmt.Sprintf(template+": #%s", order.ID),
        Data:    map[string]string{"order_id": order.ID},
    }))

    // 發送簡訊（重要事件）
    if msg.Subject == "orders.delivered" {
        s.nc.Publish("notifications.sms", json.Marshal(SMS{
            Phone:   order.CustomerPhone,
            Message: fmt.Sprintf("您的包裹 #%s 已送達！", order.ID),
        }))
    }
}
